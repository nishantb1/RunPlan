## Design

- **Older routes:**
  - At launch, `RunStore.moveEmbeddedRoutes()` re-saves every record that still holds its route, which moves the route into `routes/<id>.json`.
  - `save` moves an embedded route across when the record passed in has none. Saving a record from `all()` therefore never loses a route.
- **Auto-pause at the start:**
  - Nothing pauses until the runner has set off. Setting off uses the same test as resuming: faster than 1.2 m/s over the window, and further from the start than the two readings' combined accuracy. The start is the most accurate fix so far, the accuracy is not capped for this, and only movement while running counts.
  - A restored workout gets a fresh detector, so it also waits for the runner to set off again.
- **Checkpoint route:** `ActiveWorkoutStore` replaces the session's direct use of `active.json`.
  - The checkpoint without its route goes to `active.json` every 5 s, as now.
  - The route goes to `active-route.json`, tagged with the workout's start time, after the checkpoint. It is written only when it has grown: every 30 s, or every 5 s on forced saves (pause, resume, skip, auto-pause).
  - Loading attaches the route only when the start times match.
  - A checkpoint from an older build, with the route inside, loads as it is.
- **Route segments:**
  - `RoutePoint.startsSegment` is `true` on the first point kept after a pause and `nil` otherwise, so older routes decode unchanged.
  - `DistanceAccumulator` sets it when a pause ends: the session calls `breakRoute()` on every pause and on restore, which also makes the first fix after the pause the new reference, so nothing covered while paused is counted.
  - `[RoutePoint].segments` splits a route at those points.
  - `RouteMap` draws one line per segment of two or more points.

```swift
import Foundation

/// Run history: one file per run, so a long history never rewrites one large file. Each run's
/// route has its own file too, so listing runs never loads routes.
public struct RunStore: Sendable {
    private static let folder = "records"
    private static let routeFolder = "routes"
    private let store: JSONStore

    public init(store: JSONStore) {
        self.store = store
    }

    /// Saves a finished run with its route. The route is written first, so the record is the
    /// commit point: an interrupted save leaves at most an unlisted route file, never a listed
    /// run that has lost its route. A record from `all()` carries no route; if its file still
    /// holds one from before routes had their own files, that route is moved across, not lost.
    /// An empty route never clears a route already saved.
    public func save(_ record: RunRecord) throws {
        let route = record.route.isEmpty ? embeddedRoute(record.id) : record.route
        if !route.isEmpty {
            try store.save(route, as: routeName(record.id))
        }
        var listed = record
        listed.route = []
        try store.save(listed, as: recordName(record.id))
    }

    /// Moves routes still kept inside older records into their own files, so listing runs no
    /// longer decodes them. Cheap once every route has moved, so it runs at each launch.
    public func moveEmbeddedRoutes() {
        for name in recordNames() {
            guard let record = store.load(RunRecord.self, from: "\(Self.folder)/\(name)"),
                  !record.route.isEmpty else { continue }
            try? save(record)
        }
    }

    /// Every readable record, newest first, without routes (see `route(for:)`).
    public func all() -> [RunRecord] {
        recordNames()
            .compactMap { name -> RunRecord? in
                // Older records carry their route inside; drop it straight away.
                guard var record = store.load(RunRecord.self, from: "\(Self.folder)/\(name)") else { return nil }
                record.route = []
                return record
            }
            .sorted { $0.startedAt > $1.startedAt }
    }

    /// The run's route. Records saved before routes had their own files keep it inside until
    /// `moveEmbeddedRoutes()` runs.
    public func route(for record: RunRecord) -> [RoutePoint] {
        store.load([RoutePoint].self, from: routeName(record.id)) ?? embeddedRoute(record.id)
    }

    /// Deletes every run and route.
    public func deleteAll() {
        store.delete(Self.folder)
        store.delete(Self.routeFolder)
    }

    private func recordNames() -> [String] {
        store.names(in: Self.folder).filter { $0.hasSuffix(".json") }
    }

    /// The route inside a record's own file, from before routes had their own files.
    private func embeddedRoute(_ id: UUID) -> [RoutePoint] {
        store.load(RunRecord.self, from: recordName(id))?.route ?? []
    }

    private func recordName(_ id: UUID) -> String {
        "\(Self.folder)/\(id.uuidString).json"
    }

    private func routeName(_ id: UUID) -> String {
        "\(Self.routeFolder)/\(id.uuidString).json"
    }
}
```

```swift
import Foundation

/// Decides when an outdoor run should pause because the runner has stopped, and resume when
/// they set off again, from straight-line speed over the last 6 seconds of fixes. Straight line
/// is used because jitter adds to a path length; the cost is that a very sharp turnaround can
/// pause briefly, which corrects itself a few seconds later.
public struct AutoPauseDetector: Codable, Equatable, Sendable {
    public enum Action: String, Codable, Sendable {
        case pause
        case resume
    }

    static let windowSeconds: TimeInterval = 6
    static let minimumSpanSeconds: TimeInterval = 4
    /// Slower than this while running means the runner has stopped.
    static let stoppedMetersPerSecond = 0.5
    /// Faster than this after an automatic pause means they have set off again. The gap between
    /// the two thresholds stops GPS jitter flipping between paused and running.
    static let movingMetersPerSecond = 1.2
    /// The most a resume must move away from where the runner stopped: their two readings'
    /// accuracy, capped here, so GPS wander while standing still never resumes the run.
    static let maxNoiseFloorMeters = 15.0

    private var fixes: [LocationSample] = []
    /// Where the runner stopped. Measuring the resume from here, not across the 6 s window,
    /// lets a walk resume too: at walking pace the window alone never clears the accuracy.
    private var stoppedAt: LocationSample?
    /// Where the runner is waiting to start: the most accurate fix so far, until they have
    /// clearly set off from it. Nothing pauses before then, since GPS wanders while it settles
    /// and the runner may stand waiting for it. A slow walk (1.2 m/s or less) does not count.
    private var start: LocationSample?
    /// Set once, by running away from `start`; never cleared.
    private var hasSetOff = false
    public private(set) var isAutoPaused = false

    public init() {}

    /// Feed every fix that passed `SampleFilter`. `isRunning` is whether the workout is running
    /// (not paused). Returns the change to make, if any.
    public mutating func update(_ sample: LocationSample, isRunning: Bool) -> Action? {
        fixes.append(sample)
        fixes.removeAll { sample.timestamp.timeIntervalSince($0.timestamp) > Self.windowSeconds }
        if !hasSetOff, start.map({ sample.horizontalAccuracy < $0.horizontalAccuracy }) ?? true {
            start = sample
        }
        guard let first = fixes.first else { return nil }
        let span = sample.timestamp.timeIntervalSince(first.timestamp)
        guard span >= Self.minimumSpanSeconds else { return nil }
        let speed = Geo.distance(first, sample) / span

        if !hasSetOff {
            // Only while running: moving while paused by hand (say, walking to the route) is not
            // setting off. The accuracy is not capped here, so a poor early fix cannot open it.
            if isRunning, let start, Self.hasMovedOn(from: start, to: sample, speed: speed, capped: false) {
                hasSetOff = true
                self.start = nil
            }
            return nil
        }
        if isRunning, !isAutoPaused, speed < Self.stoppedMetersPerSecond {
            isAutoPaused = true
            stoppedAt = sample
            return .pause
        }
        if isAutoPaused, let stoppedAt, Self.hasMovedOn(from: stoppedAt, to: sample, speed: speed, capped: true) {
            isAutoPaused = false
            self.stoppedAt = nil
            return .resume
        }
        return nil
    }

    /// Call when the runner pauses or resumes by hand. A manual pause is then never resumed
    /// automatically, and after a manual resume the detector waits for fresh fixes rather than
    /// judging the runner by the seconds they spent standing still.
    public mutating func manualOverride() {
        isAutoPaused = false
        stoppedAt = nil
        fixes.removeAll()
    }

    /// Faster than `movingMetersPerSecond` over the window, and further from `anchor` than the
    /// two readings' accuracy (`capped` at `maxNoiseFloorMeters`), so GPS wander while standing
    /// never counts as moving on.
    private static func hasMovedOn(from anchor: LocationSample, to sample: LocationSample, speed: Double, capped: Bool) -> Bool {
        guard speed > movingMetersPerSecond else { return false }
        let accuracy = anchor.horizontalAccuracy + sample.horizontalAccuracy
        let noiseFloor = capped ? min(accuracy, maxNoiseFloorMeters) : accuracy
        return Geo.distance(anchor, sample) > noiseFloor
    }
}
```

```swift
import Foundation

/// The crash checkpoint for a workout in progress. The route grows by about a point a second,
/// so it has its own file, written less often than the small checkpoint of clocks and totals.
/// A restored run's distance is as current as the last checkpoint; its route may end earlier,
/// by up to the interval the caller saves the route at, and is never ahead of the distance.
public struct ActiveWorkoutStore: Sendable {
    static let file = "active.json"
    static let routeFile = "active-route.json"
    private let store: JSONStore

    public init(store: JSONStore) {
        self.store = store
    }

    /// Writes the checkpoint without its route, then the route to its own file as well when
    /// `includingRoute` is true. The checkpoint goes first, so a failed or interrupted route
    /// write never costs the checkpoint.
    public func save(_ active: ActiveWorkout, includingRoute: Bool) throws {
        guard let outdoor = active.outdoor else {
            try store.save(active, as: Self.file)
            return
        }
        var small = active
        var stripped = outdoor
        stripped.replaceRoute([])
        small.outdoor = stripped
        try store.save(small, as: Self.file)
        if includingRoute {
            try store.save(SavedRoute(startedAt: active.startedAt, points: outdoor.route), as: Self.routeFile)
        }
    }

    /// The unfinished workout with the route last saved for it, or nil when there is none.
    public func load() -> ActiveWorkout? {
        guard var active = store.load(ActiveWorkout.self, from: Self.file) else { return nil }
        // Checkpoints written before the route had its own file still hold it inside.
        if var outdoor = active.outdoor, outdoor.route.isEmpty,
           let saved = store.load(SavedRoute.self, from: Self.routeFile),
           saved.startedAt == active.startedAt {
            outdoor.replaceRoute(saved.points)
            active.outdoor = outdoor
        }
        return active
    }

    public func delete() {
        store.delete(Self.file)
        store.delete(Self.routeFile)
    }

    /// The route on disk, tagged with its workout's start so a file left from another workout
    /// is never attached to this one.
    private struct SavedRoute: Codable {
        var startedAt: Date
        var points: [RoutePoint]
    }
}
```

import Foundation

extension Array where Element == RoutePoint {
    /// The route split at each pause, so the map draws no line across ground covered while
    /// paused. Routes saved before pauses were marked come back as one segment. A GPS gap with
    /// no pause does not split it: that distance was counted, so the line shows it.
    public var segments: [[RoutePoint]] {
        var result: [[RoutePoint]] = []
        for point in self {
            if point.startsSegment == true || result.isEmpty {
                result.append([point])
            } else {
                result[result.count - 1].append(point)
            }
        }
        return result
    }

    /// True when some segment has two points or more, so the map has a line to draw.
    public var hasLine: Bool {
        segments.contains { $0.count > 1 }
    }
}
